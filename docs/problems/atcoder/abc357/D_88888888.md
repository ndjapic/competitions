# Problem: D_88888888.pas

```pascal
program D_88888888;
const
    prime = 998244353;
var
    n, x: int64;
    p, ans: int32;

function modpow(b, e: int32): int32;
begin
    if e = 0 then
        modpow := 1
    else if odd(e) then
        modpow := int64(b) * modpow(b, e-1) mod prime
    else
        modpow := modpow(int64(b) * b mod prime, e div 2);
end;

begin
    readln(n);
    p := 1;
    x := n;
    while x > 0 do begin
        p := p * 10 mod prime;
        x := x div 10;
    end;

    ans := modpow(p, n mod (prime - 1));
    ans := (ans - 1 + prime) * modpow(p - 1 + prime, prime - 2) mod prime;
    ans := n mod prime * ans mod prime;
    writeln(ans);
end.
```
