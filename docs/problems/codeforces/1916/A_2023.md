# Problem: A_2023.pas

```pascal
program A_2023;
const
    maxn = 5;
var
    ntc, tci: int8;
    n, k, i: int8;
    bi: int16;
    p: int64;

begin
    readln(ntc);
    for tci := 1 to ntc do begin

		readln(n, k);

        p := 1;
        for i := 1 to n do begin
            read(bi);
            p := p * bi;
        end;
        readln;

        if 2023 mod p > 0 then
            writeln('NO')
        else begin
            writeln('YES');
            for i := 1 to k-1 do write('1 ');
            writeln(2023 div p);
        end;

    end;
end.
```
