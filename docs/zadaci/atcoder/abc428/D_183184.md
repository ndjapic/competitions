program E_Farthest_Vertex;
const
	nn = 500 * 1000;
var
	n, i, a, b: int32;
	adj: array [1 .. nn] of array of int32;
	par, vert, dist: array [1 .. nn] of int32;

procedure addedge(a, b: int32);
begin
	if length(adj[a]) = deg[a] then setlength(adj[a], 2*deg[a]);
	adj[a][deg[a]] := b;
	inc(deg[a]);
end;

procedure dfs(a: int32);
var
	i, b, d, v: int32;
begin
	d := -1;
	for i := 0 to deg[a] - 1 do begin
		b := adj[a][i];
		if par[a] <> b then begin
			par[b] := a;
			dfs(b);
		end;
	end;
end;

begin
	readln(n);

	for a := 1 to n do begin
		setlength(adj[a], 1);
		deg[a] := 0;
		f[a] := a;
		d[a] := 0;
	end;

	for i := 1 to n-1 do begin
		readln(a, b);
		addedge(a, b);
		addedge(b, a);
	end;

	par[1] := 0;
	dfs(1);
end.
